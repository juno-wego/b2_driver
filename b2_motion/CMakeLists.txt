cmake_minimum_required(VERSION 3.16)
project(b2_motion)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(b2_interface REQUIRED)
add_executable(b2_posture_adapter src/b2_posture_adapter.cpp)
target_compile_features(b2_posture_adapter PUBLIC cxx_std_17)
ament_target_dependencies(b2_posture_adapter rclcpp std_srvs b2_interface)
install(TARGETS b2_posture_adapter DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME posture_runtime COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_posture_runtime.py $<TARGET_FILE:b2_posture_adapter>)
  set_tests_properties(posture_runtime PROPERTIES TIMEOUT 30)
endif()
ament_package()
